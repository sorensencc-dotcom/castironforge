// Environment variables
const {
  GOOGLE_CLIENT_ID,
  GOOGLE_CLIENT_SECRET,
  GOOGLE_REDIRECT_URI,
  GOOGLE_REFRESH_TOKEN,
  GOOGLE_PROJECT_ROOT_FOLDER_ID,
} = process.env;

// OAuth client
function getOAuth2Client() {
  const oAuth2Client = new google.auth.OAuth2(
    GOOGLE_CLIENT_ID,
    GOOGLE_CLIENT_SECRET,
    GOOGLE_REDIRECT_URI
  );

  if (GOOGLE_REFRESH_TOKEN) {
    oAuth2Client.setCredentials({ refresh_token: GOOGLE_REFRESH_TOKEN });
  } else {
    const tokenPath = path.join(__dirname, "tokens.json");
    if (fs.existsSync(tokenPath)) {
      const tokens = JSON.parse(fs.readFileSync(tokenPath, "utf8"));
      if (tokens.refresh_token) {
        oAuth2Client.setCredentials({ refresh_token: tokens.refresh_token });
      }
    }
  }

  return oAuth2Client;
}

// ---------- OAuth flow ----------
app.get("/auth", (req, res) => {
  const oAuth2Client = getOAuth2Client();

  const authUrl = oAuth2Client.generateAuthUrl({
    access_type: "offline",
    scope: ["https://www.googleapis.com/auth/drive"],
    prompt: "consent",
  });

  res.redirect(authUrl);
});

app.get("/oauth2callback", async (req, res) => {
  try {
    const code = req.query.code;
    const oAuth2Client = getOAuth2Client();

    const { tokens } = await oAuth2Client.getToken(code);
    const tokenPath = path.join(__dirname, "tokens.json");
    fs.writeFileSync(tokenPath, JSON.stringify(tokens, null, 2), "utf8");

    res.send("Google Drive MCP server authenticated. You can close this window.");
  } catch (err) {
    console.error("[oauth2callback] Error:", err);
    res.status(500).send("Authentication failed.");
  }
});

app.get("/catalog/folder-map", async (req, res) => {

// ---------- Catalog: consolidate duplicates ----------
app.post("/catalog/consolidate-duplicates", async (req, res) => {
  try {
    const auth = getOAuth2Client();
    const { folderId, mode } = req.body;

    const targetFolderId = folderId || GOOGLE_PROJECT_ROOT_FOLDER_ID;
    const files = await listFiles(auth, targetFolderId);
    const duplicates = await findDuplicates(auth, files);

    const actions = await consolidateDuplicates(auth, duplicates, mode || "trash-duplicates");
    res.json({ actions });
  } catch (err) {
    console.error("[consolidate-duplicates] Error:", err);
    res.status(500).json({ error: "Failed to consolidate duplicates" });
  }
});

app.get("/catalog/classify-file", async (req, res) => {
  try {
    const auth = getOAuth2Client();
    const fileId = req.query.fileId;

    if (!fileId) return res.status(400).json({ error: "fileId is required" });

// ---------- Catalog: move file ----------
app.post("/catalog/move-file", async (req, res) => {
  try {
    const auth = getOAuth2Client();
    const { fileId, destinationFolderId } = req.body;

    if (!fileId || !destinationFolderId) {
      return res.status(400).json({ error: "fileId and destinationFolderId are required" });
    }

    const result = await moveFile(auth, fileId, destinationFolderId);
    res.json(result);
  } catch (err) {
    console.error("[move-file] Error:", err);
    res.status(500).json({ error: "Failed to move file" });
  }
});

// ---------- Catalog: bulk move ----------
app.post("/catalog/bulk-move", async (req, res) => {
  try {
    const auth = getOAuth2Client();
    const { moves } = req.body;

    if (!Array.isArray(moves) || moves.length === 0) {
      return res.status(400).json({ error: "moves must be a non-empty array" });
    }

    const results = await bulkMove(auth, moves);
    res.json({ results });
  } catch (err) {
    console.error("[bulk-move] Error:", err);
    res.status(500).json({ error: "Failed to perform bulk move" });
  }
});

// ---------- Start server ----------
const PORT = process.env.PORT || 3001;
app.listen(PORT);
