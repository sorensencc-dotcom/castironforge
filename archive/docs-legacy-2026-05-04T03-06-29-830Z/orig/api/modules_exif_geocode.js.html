/**
 * geocode.cjs — Version 1.0.0
 * Added: 2026-03-31
 *
 * Description:
 * - New EXIF GPS → Reverse Geocoding module (CommonJS)
 * - Converts DMS → decimal degrees
 * - Uses OpenStreetMap Nominatim for reverse geocoding
 * - Provides extractLocationFromExif() for ingest pipeline integration
 * - Returns structured location metadata (city, state, country, confidence)
 */

const fetch = require("node-fetch");

/**
 * Convert EXIF DMS format to decimal degrees.
 */
function dmsToDecimal(dms, ref) {
  if (!dms || !Array.isArray(dms) || dms.length !== 3) return null;

  const [degrees, minutes, seconds] = dms.map(Number);
  let decimal = degrees + minutes / 60 + seconds / 3600;

  if (ref === "S" || ref === "W") {
    decimal = decimal * -1;
  }

  return decimal;
}

/**
 * Reverse geocode using OpenStreetMap Nominatim.
 */
async function reverseGeocode(lat, lon) {
  const url = `https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}`;

  const response = await fetch(url, {
    headers: {
      "User-Agent": "CastIronCharlie-MCP/1.0 (contact: your-email@example.com)"
    }
  });

  if (!response.ok) {
    throw new Error(`Geocoding failed: ${response.status}`);
  }

  const data = await response.json();

  return {
    latitude: lat,
    longitude: lon,
    city: data.address.city || data.address.town || data.address.village || null,
    state: data.address.state || null,
    country: data.address.country || null,
    displayName: data.display_name || null,
    confidence: data.importance || 0.5
  };
}

/**
 * Extract GPS from EXIF and return location metadata.
 */
async function extractLocationFromExif(exif) {
  if (!exif) return null;

  const lat = dmsToDecimal(exif.GPSLatitude, exif.GPSLatitudeRef);
  const lon = dmsToDecimal(exif.GPSLongitude, exif.GPSLongitudeRef);

  if (!lat || !lon) return null;

  return await reverseGeocode(lat, lon);
}

module.exports = {
  dmsToDecimal,
  reverseGeocode,
  extractLocationFromExif
};
